CREATE TABLE Schedule(
	schedule_id INT PRIMARY KEY,
	bus_id INT,
    route_id INT,
    departure_time TIME,
    travel_date DATE,
    
    FOREIGN KEY (bus_id) REFERENCES Bus(bus_id),
    FOREIGN KEY (route_id) REFERENCES Route(route_id)
	);

CREATE TABLE Ticket(
	ticket_id INT PRIMARY KEY,

    FOREIGN KEY (schedule_id) REFERENCES Schedule(schedule_id)
    );

#what join is doing #basic example.
SELECT passenger.name, ticket.seat_number
FROM ticket
JOIN passenger
ON ticket.passenger_id = passenger.passenger_id;



SELECT 
    Passenger.name,
    Bus.bus_number,
    Route.source,
    Route.destination,
    Schedule.travel_date,
    Schedule.departure_time,
    Ticket.seat_number,
    Ticket.status
FROM Ticket
JOIN Passenger ON Ticket.passenger_id = Passenger.passenger_id
JOIN Schedule ON Ticket.schedule_id = Schedule.schedule_id
JOIN Bus ON Schedule.bus_id = Bus.bus_id
JOIN Route ON Schedule.route_id = Route.route_id;

#Count booked seats
SELECT COUNT(*) AS booked_seats
FROM Ticket
WHERE schedule_id = 1
AND status = 'book';


#Find remaining seats
SELECT 
    Bus.total_seats - COUNT(Ticket.ticket_id) AS available_seats
FROM Schedule
JOIN Bus ON Schedule.bus_id = Bus.bus_id
LEFT JOIN Ticket ON Schedule.schedule_id = Ticket.schedule_id
AND Ticket.status = 'book'
WHERE Schedule.schedule_id = 1
GROUP BY Bus.total_seats;
